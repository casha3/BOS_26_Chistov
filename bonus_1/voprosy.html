1/*
2 * bb_cov_runtime.c
3 *
4 * Runtime component for collecting "Basic Block Coverage" of C/C++
5 * programs compiled with Clang/LLVM SanitizerCoverage.
6 *
7 * Instrumentation flags (see task statement):
8 *
9 * -fprofile-instr-generate
10 * -fcoverage-mapping
11 * -fsanitize-coverage=bb,no-prune,inline-bool-flag,pc-table
12 *
13 * The compiler inserts an inline boolean flag for every basic block and
14 * calls
15 *
16 * __sanitizer_cov_bool_flag_init(bool *start, bool *end)
17 *
18 * once per module. The array [start, end) contains one flag per
19 * instrumented block; a flag is set to true iff the corresponding block
20 * was executed at least once (inline-bool-flag).
21 *
22 * Together with pc-table the compiler also provides the table of PCs of
23 * all instrumented blocks:
24 *
25 * __sanitizer_cov_pcs_init(const uintptr_t *pcs_beg,
26 * const uintptr_t *pcs_end)
27 *
28 * The table contains pairs [PC, PCFlags] -- one pair per instrumented
29 * block. The number of pairs equals the number of boolean flags, which
30 * lets us count the total number of instrumented basic blocks and, at
31 * process exit, the number of executed blocks.
32 *
33 * Atexit the runtime writes a raw JSON report containing every PC plus
34 * the indices of executed blocks (union of ALL runs, so that a basic
35 * block executed by several test runs is counted only once -- this is
36 * achieved by merging the previously existing report on each run).
37 *
38 * The raw report is turned into a per-source-file report by
39 * tools/bb_cov_finalize.py, which resolves every PC to a file/line
40 * with llvm-addr2line.
41 */

76/*
77 * Called once per DSO during its construction with the array of inline
78 * boolean flags, one per instrumented basic block.
79 */
80void __sanitizer_cov_bool_flag_init(bool *start, bool *end)
81{
82 size_t i;
83
84 if (start == NULL || start == end)
85 return;
86
87 /* A given flag array may be reported more than once (LLVM: "may be
88 * called more than once with the same values"); register it only
89 * once. */
90 for (i = 0; i < g_module_count; i++) {
91 if (g_modules[i].flags == (uint8_t *)start)
92 return;
93 }
94 if (g_module_count >= BBCOV_MAX_MODULES)
95 return;
96
97 bb_module_t *m = &g_modules[g_module_count];
98 m->flags = (uint8_t *)start;
99 m->num = (size_t)(end - start); /* bool has size 1 byte */
100 m->pcs = NULL;
101 m->pcs_count = 0;
102
103 g_total_blocks += m->num;
104 g_module_count++;
105}

107/*
108 * Called once per DSO with the PC table: pairs [PC, PCFlags], one pair
109 * per instrumented block. We copy the PCs because the table is owned by
110 * the compiler and its lifetime is not guaranteed.
111 *
112 * Note: SanitizerCoverage reports the bool flags and the PC table as two
113 * independent callbacks, so we pair them up here: every module (flag
114 * array) without an assigned PC table receives the next table.
115 */
116void __sanitizer_cov_pcs_init(const uintptr_t *pcs_beg,
117 const uintptr_t *pcs_end)
118{
119 size_t i;
120
121 if (pcs_beg == NULL || pcs_beg == pcs_end)
122 return;
123
124 size_t entries = (size_t)(pcs_end - pcs_beg) / 2;
125
126 for (i = 0; i < g_module_count; i++) {
127 if (g_modules[i].pcs == NULL) {
128 bb_module_t *m = &g_modules[i];
129 uintptr_t *pcs;
130
131 if (entries < m->num)
132 entries = m->num;
133 pcs = (uintptr_t *)calloc(entries, sizeof(uintptr_t));
134 if (pcs == NULL)
135 return;
136
137 for (size_t k = 0; k < entries; k++)
138 pcs[k] = (k < entries) ? pcs_beg[2 * k] : 0;
139
140 m->pcs = pcs;
141 m->pcs_count = entries;
142 return;
143 }
144 }
145}

172/*
173 * Merge the executed indices stored in a previously produced raw report
174 * into the bitmask EXEC (union). This guarantees that a basic block
175 * executed by several different test runs is counted only once.
176 */
177static void bb_merge_previous(const char *path, uint8_t *exec, size_t n)
178{
179 FILE *f = fopen(path, "r");
180 char buf[4096];
181 char *p;
182 char *q;
183
184 if (f == NULL)
185 return;
186
187 while (fgets(buf, sizeof(buf), f) != NULL) {
188 /* Find the "executed_indices" field within the current buffer
189 * (the field fits in one line in our own format). */
190 p = strstr(buf, "\"executed_indices\"");
191 if (p == NULL)
192 continue;
193 p = strchr(p, '[');
194 if (p == NULL)
195 continue;
196 p++;
197 while (*p && *p != ']') {
198 while (*p && (*p == ' ' || *p == ',' || *p == '\t'))
199 p++;
200 if (!*p || *p == ']')
201 break;
202 unsigned long idx = strtoul(p, &q, 10);
203 if (idx < n)
204 exec[idx] = 1;
205 p = q;
206 }
207 break;
208 }
209 fclose(f);
210}

276/*
277 * Atexit handler: compute this-run execution from the bool flags, merge
278 * with previously recorded runs and dump the raw JSON report.
279 */
280static void bb_cov_dump(void)
281{
282 size_t i;
283 size_t base = 0;
284 uint8_t *exec;
285 size_t executed = 0;
286
287 if (g_total_blocks == 0)
288 return;
289
290 exec = (uint8_t *)calloc(g_total_blocks, 1);
291 if (exec == NULL)
292 return;
293
294 /* This-run executed blocks. */
295 for (i = 0; i < g_module_count; i++) {
296 bb_module_t *m = &g_modules[i];
297 size_t k;
298
299 for (k = 0; k < m->num; k++) {
300 if (m->flags[k])
301 exec[base + k] = 1;
302 }
303 base += m->num;
304 }
305
306 /* Union with the previous run(s) to avoid double counting. */
307 bb_merge_previous(g_out_name, exec, g_total_blocks);
308
309 for (i = 0; i < g_total_blocks; i++)
310 executed += exec[i];
311
312 bb_write_json(g_out_name, exec, g_total_blocks);
313
314 if (!g_quiet)
315 fprintf(stderr,
316 "bb_cov_runtime: %zu instrumented basic blocks, "
317 "%zu executed\n",
318 g_total_blocks, executed);
319
320 free(exec);
321}

80Зачем функция __sanitizer_cov_bool_flag_init и что она сохраняет?Компилятор SanitizerCoverage (режим inline-bool-flag) для каждого базового блока вставляет установку inline-флага и вызывает этот колбэк с диапазоном флагов [start, end) этого DSO — по одному флагу на блок. Runtime сохраняет указатель на массив флагов, так как по выставленному флагу можно определить число выполненных блоков.
90Почему нужна проверка "register it only once"?LLVM может вызвать колбэк более одного раза с одинаковыми значениями для одного модуля. Если повторно приходит тот же адрес массива флагов, runtime выходит сразу, чтобы не задвоить число инструментированных блоков (g_total_blocks).
116Что делает __sanitizer_cov_pcs_init и что такое PC-таблица?Получает таблицу пар [PC, PCFlags] — по одной паре на инструментированный блок. Число пар равно числу флагов (это и есть общее число блоков). Runtime копирует адреса в собственную память, потому что таблица принадлежит компилятору и её время жизни не гарантировано; PC используются потом для привязки блоков к файлам через llvm-addr2line.
280Что делает обработчик bb_cov_dump, регистрируемый через atexit?На выходе процесса строит битовую маску выполненных блоков из inline-флагов, объединяет её с данными предыдущих прогонов (bb_merge_previous — union, чтобы базовый блок, исполненный несколькими тестами, считался один раз) и записывает сырой JSON (адреса всех блоков + индексы выполненных).
327Зачем конструктор bb_cov_init (__attribute__((constructor)))?При загрузке программы читает переменные окружения BB_COV_OUT (куда писать отчёт) и BB_COV_QUIET и регистрирует atexit(bb_cov_dump), чтобы отчёт формировался автоматически при завершении тестового запуска.

56Зачем функция normaddr?Приводит адрес к каноническому виду 0x%x (без ведущих нулей). В raw-отчёте runtime и в выводе llvm-addr2line число шестнадцатеричных цифр может отличаться (0x0000000ad0 vs 0xad0), поэтому без нормализации сопоставить PC с файлом было бы нельзя.
64Как resolve_pcs связывает адрес базового блока с исходным файлом?Одним bulk-запуском llvm-addr2line -e <bin> -f -a -C <все PC> адреса резолвятся в «файл:строку». Выход парсится строго по 3 строки на адрес (адрес, имя функции, расположение); опция -i не используется, чтобы не появлялись дополнительные кадры встраивания.
136Как считаются «найдено/выполнено» блоков по файлу?Для каждого PC определяется файл строки; блок приписывается этому файлу (found++), а если его индекс есть в executed_indices, то и hit++. На выходе получается пофайловая карта {файл: {found, hit}} — данные для отчёта genhtml.
153Что делает итоговая сводка total_found/total_hit?Суммирует found и hit по всем файлам и кладёт в отчёт поля resolved_found/resolved_hit; они отражают реально приписанные к файлам блоки, что позволяет сравнить с общим числом инструментированных блоков и обнаружить потери при резолвинге адресов.
178Почему появляется предупреждение WARNING?Если суммарное число блоков, приписанных файлам, заметно меньше общего числа инструментированных блоков (менее 10%), значит PC не резолвятся — как правило, бинарник собран PIE (нет -no-pie) и адреса сдвинуты ASLR. Это подсказка проверить флаг -no-pie и -g.

64# --- llvm-cov export (handles different CLI versions) ---------------------
65# `llvm-cov export` changed its option style over the years:
66# LLVM < 18: llvm-cov export -instr-profile=P -format=lcov BIN > out
67# LLVM >= 18: llvm-cov export --instr-profile=P --format=lcov BIN > out
68# We simply try both forms.
69
70llvm_cov_export_lcov() {
71 local bin="$1"
72 local profdata="$2"
73 local out="$3"
74
75 rm -f "$out"
76
77 if "$LLVM_COV" export --format=lcov --instr-profile="$profdata" "$bin" \
78 > "$out" 2>/dev/null; then
79 :
80 elif "$LLVM_COV" export -format=lcov -instr-profile="$profdata" "$bin" \
81 > "$out" 2>/dev/null; then
82 :
83 else
84 echo "ERROR: llvm-cov export failed for $bin" >&2
85 return 1
86 fi
87
88 if [ ! -s "$out" ]; then
89 echo "ERROR: llvm-cov export produced empty output: $out" >&2
90 return 1
91 fi
92 return 0

116# --- tree_diff_identical <dirA> <dirB> ------------------------------------
117# Compare two HTML report trees ignoring the "Date" header cell (genhtml
118# writes the current time to the second into every page, so two runs that
119# straddle a second boundary would otherwise look different even though
120# the coverage data is identical). Returns 0 if the trees are identical
121# modulo that timestamp, 1 otherwise.
122tree_diff_identical() {
123 local a="$1" b="$2"
124 local da db f res=0
125
126 [ -d "$a" ] || return 1
127 [ -d "$b" ] || return 1
128
129 da=$( (cd "$a" && find . -type f | sort) )
130 db=$( (cd "$b" && find . -type f | sort) )
131 [ "$da" = "$db" ] || return 1
132
133 while IFS= read -r f; do
134 [ -n "$f" ] || continue
135 # Сравниваем только текстовые HTML/CSS (PNG генерируются детерминированно)
136 case "$f" in
137 *.html|*.css) ;;
138 *) continue ;;
139 esac
140 if ! diff <(grep -v 'headerValue">2' "$a/$f" 2>/dev/null) \
141 <(grep -v 'headerValue">2' "$b/$f" 2>/dev/null) \
142 >/dev/null 2>&1; then
143 res=1
144 fi
145 done <<< "$da"
146
147 return $res
148}

1#!/usr/bin/env bash
2#
3# build_project.sh - compile an open-source project with the combined
4# (standard + basic block) instrumentation, run its test inputs, merge
5# the resulting profiles and produce:
6#
7# output/<PROJECT_NAME>/coverage.info (line/func/branch coverage)
8# output/<PROJECT_NAME>/bb_cov.json (basic block coverage)
9#
10# The project is described by "<project_dir>/build.conf":
11#
12# PROJECT_C_FILES="..." project .c files to instrument
13# HARNESS_C_FILE="..." test driver, also instrumented
14# EXTRA_INCLUDES="-I..." include flags for project headers
15# EXTRA_LDFLAGS="..."
16# TESTS=("..." ...) one test input per array element; each
17# produces a .profraw and contributes to the
18# merged bb_cov_raw.json

54run_with_timeout() {
55 local secs="${1:-30}"
56 shift
57 local rc=0
58
59 "$@" &
60 local pid=$!
61 ( sleep "$secs"; kill -9 "$pid" 2>/dev/null ) &
62 local killer=$!
63
64 wait "$pid" 2>/dev/null; rc=$?
65 kill "$killer" 2>/dev/null
66 wait "$killer" 2>/dev/null
67 return $rc
68}

70run_tests() {
71 local i=0
72 local testname
73 local profraw
74
75 mkdir -p "$OUT_DIR/run_work"
76
77 for testname in "${TESTS[@]:-}"; do
78 i=$((i + 1))
79 profraw="$RUNS_DIR/run_$i.profraw"
80 if [ -n "$testname" ]; then

82 else

84 fi

86 # merged automatically by the runtime (union -> no double count).
87 # Tests run in a dedicated scratch directory (the microtar tests
88 # create/read a tar file there). RUN_TIMEOUT protects against
89 # programs that do not terminate on their own.
90 if [ -n "${RUN_TIMEOUT:-}" ]; then
91 ( cd "$OUT_DIR/run_work" && \
92 LLVM_PROFILE_FILE="$profraw" \
93 BB_COV_OUT="$RAW_JSON" \
94 BB_COV_QUIET=1 \
95 run_with_timeout "$RUN_TIMEOUT" "$BIN" $testname ) || true
96 else
97 ( cd "$OUT_DIR/run_work" && \
98 LLVM_PROFILE_FILE="$profraw" \
99 BB_COV_OUT="$RAW_JSON" \
100 BB_COV_QUIET=1 \
101 "$BIN" $testname ) || true
102 fi
103 done
104}

117}

54Что делает run_with_timeout?Портативный таймаут: запускает команду в фоне и через заданное число секунд убивает её (SIGKILL). Защищает от программ, которые не завершаются сами (например, fifo_reader/fifo_writer, signalwatch).
70Как прогоняются тесты и объединяются данные нескольких прогонов?Каждый тест (элемент TESTS) запускается со своим LLVM_PROFILE_FILE (отдельный .profraw) и общим BB_COV_OUT; внутри runtime выполняется union выполненного множества блоков. Таким образом несколько тестовых входов сливаются в одно покрытие без двойного счёта блоков.
106Что делает merge_profiles и как формируется coverage.info?llvm-profdata merge -sparse объединяет все .profraw в default.profdata, после чего llvm_cov_export_lcov формирует coverage.info в формате LCOV для genhtml.
119Что делает finalize_bb?Запускает bb_cov_finalize.py, который по сырому отчёту bb_cov_raw.json и бинарнику получает пофайловый УН bb_cov.json (привязка блоков к файлам включена).

121Что происходит при явном --main без --sources?Включается режим «одна программа»: компилируется только указанный файл (SOURCES пуст). Если бы источники подбирались автоматически, подхватились бы все остальные файлы с main() и сборка упала бы на повторных определениях функции main.
